# BeamNG RTSS Profile Installer
# This ONLY applies to BeamNG.drive v0.39.2.x
# RUN AS ADMIN

Write-Host ""
Write-Host "======================================================" -ForegroundColor Cyan
Write-Host " BeamNG.drive RTSS Profile Installer"
Write-Host " For BeamNG.drive v0.39.2.x ONLY"
Write-Host "======================================================" -ForegroundColor Cyan
Write-Host ""

# Check for Administrator privileges
$isAdmin = ([Security.Principal.WindowsPrincipal] `
    [Security.Principal.WindowsIdentity]::GetCurrent()).IsInRole(`
    [Security.Principal.WindowsBuiltInRole]::Administrator)

if (-not $isAdmin) {
    Write-Host "ERROR: Please run this script as Administrator." -ForegroundColor Red
    Pause
    exit
}

function Ask-YesNo($question)
{
    do {
        $answer = Read-Host "$question (Y/N)"
    } until ($answer -match '^[YyNn]$')

    return $answer -match '^[Yy]$'
}

Write-Host ""
Write-Host "Before continuing, please verify the following."
Write-Host ""

if (-not (Ask-YesNo "1. Are you using the Vulkan renderer (NOT DirectX 12)?")) {
    Write-Host "Installation cancelled."
    Pause
    exit
}

if (-not (Ask-YesNo "2. Are you using RivaTuner Statistics Server (RTSS), and does BeamNG crash immediately on launch?")) {
    Write-Host "Installation cancelled."
    Pause
    exit
}

if (-not (Ask-YesNo "3. Did you verify beamng.log contains '[ErrorDeviceLost]' errors?")) {
    Write-Host "Installation cancelled."
    Pause
    exit
}

if (-not (Ask-YesNo "4. Would you like to install the BeamNG RTSS profile?")) {
    Write-Host "Installation cancelled."
    Pause
    exit
}

$sourceFolder = Split-Path -Parent $MyInvocation.MyCommand.Path
$destination = "C:\Program Files (x86)\RivaTuner Statistics Server\ProfileTemplates"

if (!(Test-Path $destination)) {
    Write-Host ""
    Write-Host "ERROR: RTSS ProfileTemplates folder not found:" -ForegroundColor Red
    Write-Host $destination
    Pause
    exit
}

try {
    Copy-Item "$sourceFolder\BeamNG.drive.x64.exe.cfg" -Destination $destination -Force
    Copy-Item "$sourceFolder\BeamNG.drive.exe.cfg" -Destination $destination -Force

    Write-Host ""
    Write-Host "==============================================" -ForegroundColor Green
    Write-Host " Installation completed successfully!"
    Write-Host ""
    Write-Host "The following RTSS profiles were installed:"
    Write-Host " - BeamNG.drive.exe.cfg"
    Write-Host " - BeamNG.drive.x64.exe.cfg"
    Write-Host ""
    Write-Host "Please restart RTSS if it was already running."
    Write-Host "==============================================" -ForegroundColor Green
}
catch {
    Write-Host ""
    Write-Host "ERROR: Failed to copy one or more files." -ForegroundColor Red
    Write-Host $_.Exception.Message
}

Pause
